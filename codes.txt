Q-1 Check Age Category - child, teenager, or adult. 

int main () {
int a;
printf("enter your age:");
scanf("%d",&a);
if (a>0 && a<=12)
{
printf("you are child");
}
if (a>=13 && a<=19)
{
printf("you are teenager");
}
if (a>=20)
{
printf("you are adult");
}
return 0;
}
